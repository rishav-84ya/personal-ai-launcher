/**
 * RIVA — Rishav Intelligent Virtual Assistant | Cloudflare Worker (single file) v3.0
 * Supabase (app_state, chat_messages) <-> Kaggle Gradio model
 *
 * Variables: SUPABASE_URL, SUPABASE_SECRET_KEY, STATE_ID (main_ai),
 *            GRADIO_USERNAME / GRADIO_PASSWORD (optional)
 *
 * WHY THE OLD UI WAS DEAD: the page lived inside a JS template literal, so
 * '\n' inside the client <script> turned into a real line break and broke the
 * script (syntax error = nothing clickable). This version serves the HTML via
 * String.raw so the client code reaches the browser unchanged.
 */

export default {
  async fetch(request, env) {
    const CFG = {
      stateId: env.STATE_ID || 'main_ai',
      supabaseUrl: env.SUPABASE_URL,
      supabaseKey: env.SUPABASE_SECRET_KEY || '',
      gradioUsername: env.GRADIO_USERNAME || '',
      gradioPassword: env.GRADIO_PASSWORD || '',
    };
    const url = new URL(request.url);
    if (url.pathname === '/health') return json({ status: 'ok' });
    if (url.pathname === '/v1/models' || url.pathname === '/v1/chat/completions' || request.method === 'OPTIONS') {
      try { return await handleRivaApi(request, url, CFG); }
      catch (err) { return withCors(json({ error: String(err.message || err) }, 500)); }
    }
    if (url.pathname.startsWith('/api/')) {
      try { return await handleApi(request, url, CFG); }
      catch (err) { return json({ error: String(err.message || err) }, 500); }
    }
    if (request.method !== 'GET' || (url.pathname !== '/' && url.pathname !== '/index.html')) {
      return new Response('Not found.', { status: 404 });
    }
    return new Response(HTML, {
      headers: {
        'content-type': 'text/html; charset=utf-8',
        'cache-control': 'no-store',
        'content-security-policy': "default-src 'self'; style-src 'unsafe-inline' https://fonts.googleapis.com; font-src https://fonts.gstatic.com; script-src 'unsafe-inline' https://cdn.jsdelivr.net; img-src 'self' data:; connect-src 'self'; base-uri 'none'; frame-ancestors 'none'",
        'x-content-type-options': 'nosniff',
        'referrer-policy': 'no-referrer',
      },
    });
  },
};

function json(data, status = 200) {
  return new Response(JSON.stringify(data), { status, headers: { 'content-type': 'application/json; charset=utf-8' } });
}

async function sb(CFG, path, { method = 'GET', body, headers = {} } = {}) {
  if (!CFG.supabaseUrl || !CFG.supabaseKey) throw new Error('Worker secrets SUPABASE_URL and SUPABASE_SECRET_KEY are required.');
  const r = await fetch(CFG.supabaseUrl + '/rest/v1/' + path, {
    method,
    headers: {
      apikey: CFG.supabaseKey,
      Authorization: 'Bearer ' + CFG.supabaseKey,
      'Content-Type': 'application/json',
      Prefer: method === 'GET' ? '' : 'return=representation',
      ...headers,
    },
    body: body ? JSON.stringify(body) : undefined,
  });
  if (!r.ok) {
    const t = await r.text().catch(() => '');
    throw new Error('Supabase ' + r.status + ': ' + t.slice(0, 300));
  }
  const t = await r.text();
  return t ? JSON.parse(t) : null;
}

function gradioHeaders(CFG) {
  const h = new Headers({ accept: 'application/json' });
  if (CFG.gradioUsername || CFG.gradioPassword) h.set('authorization', 'Basic ' + btoa(CFG.gradioUsername + ':' + CFG.gradioPassword));
  return h;
}

async function pingGradio(gradioUrl, CFG) {
  if (!gradioUrl) return { live: false, reason: 'no-url', url: null, latencyMs: null };
  const base = gradioUrl.replace(/\/$/, '');
  const t0 = Date.now();
  let lastStatus = null;
  for (const path of ['/info', '/gradio_api/info']) {
    try {
      const r = await fetch(base + path, { headers: gradioHeaders(CFG), signal: AbortSignal.timeout(5000) });
      lastStatus = r.status;
      if (r.ok) return { live: true, url: gradioUrl, latencyMs: Date.now() - t0 };
    } catch (_) {}
  }
  return { live: false, reason: lastStatus ? 'http-' + lastStatus : 'unreachable', url: gradioUrl, latencyMs: Date.now() - t0 };
}

function parseGradioSSE(raw) {
  let out = null;
  for (const line of String(raw || '').split(/\r?\n/)) {
    if (!line.startsWith('data:')) continue;
    const p = line.slice(5).trim();
    if (!p || p === '[DONE]') continue;
    try {
      const v = JSON.parse(p);
      if (Array.isArray(v)) { const c = v.find(x => typeof x === 'string'); if (c != null) out = c; }
      else if (typeof v === 'string') out = v;
    } catch (_) {}
  }
  return out == null ? null : String(out);
}

function withCors(response) {
  const h = new Headers(response.headers);
  h.set('access-control-allow-origin', '*');
  h.set('access-control-allow-methods', 'GET,POST,OPTIONS');
  h.set('access-control-allow-headers', 'Authorization,Content-Type');
  return new Response(response.body, { status: response.status, statusText: response.statusText, headers: h });
}

async function getCurrentUrl(CFG) {
  let state = {};
  try {
    const rows = await sb(CFG, 'app_state?select=current_url,status&id=eq.' + encodeURIComponent(CFG.stateId));
    state = (rows && rows[0]) || {};
  } catch (_) {}
  let base = String(state.current_url || '').replace(/\/$/, '');
  if (!base) {
    try {
      const rows = await sb(CFG, 'gradio_urls?select=url&id=eq.1&limit=1');
      base = String(rows?.[0]?.url || '').replace(/\/$/, '');
    } catch (_) {}
  }
  return { base, status: state.status || null };
}

// OpenAI-compatible passthrough
async function handleRivaApi(request, url, CFG) {
  if (request.method === 'OPTIONS') return withCors(new Response(null, { status: 204 }));
  if (!['GET', 'POST'].includes(request.method)) return withCors(json({ error: 'method not allowed' }, 405));
  const { base, status } = await getCurrentUrl(CFG);
  if (!base) return withCors(json({ error: { message: 'RIVA offline: app_state.current_url is empty', type: 'server_error' } }, 503));
  if (url.pathname !== '/v1/models' && url.pathname !== '/v1/chat/completions') return withCors(json({ error: 'not found' }, 404));
  const headers = new Headers();
  const ct = request.headers.get('content-type');
  if (ct) headers.set('content-type', ct);
  try {
    const up = await fetch(base + url.pathname, {
      method: request.method, headers,
      body: request.method === 'POST' ? request.body : undefined,
      signal: AbortSignal.timeout(120000),
    });
    const res = new Response(up.body, { status: up.status, statusText: up.statusText, headers: up.headers });
    res.headers.set('x-riva-upstream', base + url.pathname);
    if (status) res.headers.set('x-riva-kaggle-status', status);
    return withCors(res);
  } catch (err) {
    return withCors(json({ error: { message: 'RIVA upstream unavailable: ' + String(err.message || err), type: 'server_error' } }, 502));
  }
}

async function startGradioCall(base, message, CFG, conversationId) {
  const headers = gradioHeaders(CFG);
  headers.set('content-type', 'application/json');
  const start = await fetch(base + '/gradio_api/call/frontend_chat', {
    method: 'POST', headers,
    body: JSON.stringify({ data: [message, 'Auto', false, conversationId || 'personal-chat-v4-default'] }),
    signal: AbortSignal.timeout(20000),
  });
  if (!start.ok) throw new Error('Gradio start failed (' + start.status + ')');
  const started = await start.json().catch(() => ({}));
  if (!started.event_id) throw new Error('Gradio did not return an event id.');
  return started.event_id;
}

async function runGradioCommand(gradioUrl, message, CFG, conversationId) {
  const base = (gradioUrl || '').replace(/\/$/, '');
  if (!base) return { ok: false, error: 'No Gradio URL available.' };
  try {
    const id = await startGradioCall(base, message, CFG, conversationId);
    const up = await fetch(base + '/gradio_api/call/frontend_chat/' + encodeURIComponent(id), { headers: gradioHeaders(CFG), signal: AbortSignal.timeout(30000) });
    if (!up.ok) return { ok: false, error: 'Gradio command stream failed (' + up.status + ').' };
    return { ok: true, reply: parseGradioSSE(await up.text()) || '' };
  } catch (e) { return { ok: false, error: String(e.message || e) }; }
}

async function streamGradioToBrowser(gradioUrl, message, CFG, conversationId, modelSelection) {
  const base = (gradioUrl || '').replace(/\/$/, '');
  if (!base) return json({ error: 'RIVA is offline. Start the server from the Server tab.' }, 503);
  let eventId, upstream;
  try {
    eventId = await startGradioCall(base, message, CFG, conversationId);
    upstream = await fetch(base + '/gradio_api/call/frontend_chat/' + encodeURIComponent(eventId), { headers: gradioHeaders(CFG), signal: AbortSignal.timeout(120000) });
  } catch (e) { return json({ error: String(e.message || e) }, 502); }
  if (!upstream.ok || !upstream.body) return json({ error: 'Gradio stream failed (' + upstream.status + ')' }, 502);

  const enc = new TextEncoder(), dec = new TextDecoder(), reader = upstream.body.getReader();
  let buffer = '';
  const send = (c, o) => c.enqueue(enc.encode('data: ' + JSON.stringify(o) + '\n\n'));
  const stream = new ReadableStream({
    async start(controller) {
      try {
        send(controller, { type: 'status', status: 'connected', model: modelSelection || 'auto' });
        while (true) {
          const { value, done } = await reader.read();
          if (done) break;
          buffer += dec.decode(value, { stream: true });
          const lines = buffer.split(/\r?\n/);
          buffer = lines.pop() || '';
          for (const line of lines) {
            if (!line.startsWith('data:')) continue;
            const p = line.slice(5).trim();
            if (!p || p === '[DONE]') continue;
            let v;
            try { v = JSON.parse(p); } catch (_) { continue; }
            if (v && v.error) { send(controller, { type: 'error', message: String(v.error.message || v.error) }); continue; }
            let cand = null;
            if (Array.isArray(v)) cand = v.find(x => typeof x === 'string');
            else if (typeof v === 'string') cand = v;
            else if (v && typeof v.output === 'string') cand = v.output;
            if (cand == null) continue;
            cand = String(cand);
            send(controller, { type: 'delta', text: cand });
            const mm = cand.match(/Qwen\s+2\.5\s+(7B|14B)/i);
            if (mm) send(controller, { type: 'model', model: 'Qwen 2.5 ' + mm[1].toUpperCase() });
          }
        }
        send(controller, { type: 'done' });
        controller.close();
      } catch (e) {
        try { send(controller, { type: 'error', message: String(e.message || e) }); } catch (_) {}
        try { controller.close(); } catch (_) {}
      } finally { try { reader.releaseLock(); } catch (_) {} }
    },
    cancel() { try { reader.cancel(); } catch (_) {} },
  });
  return new Response(stream, { status: 200, headers: { 'content-type': 'text/event-stream; charset=utf-8', 'cache-control': 'no-cache, no-transform', 'x-accel-buffering': 'no' } });
}

async function handleApi(request, url, CFG) {
  const m = request.method, p = url.pathname;
  const stateRow = async () => {
    const rows = await sb(CFG, 'app_state?select=current_url,desired_state,status&id=eq.' + encodeURIComponent(CFG.stateId));
    return (rows && rows[0]) || {};
  };

  if (p === '/api/state' && m === 'GET') {
    const state = await stateRow();
    return json({ state, ping: await pingGradio(state.current_url, CFG) });
  }

  if (p === '/api/power' && m === 'POST') {
    const body = await request.json().catch(() => ({}));
    const desired = body.desired_state === 'offline' ? 'offline' : 'online';
    const rows = await sb(CFG, 'app_state?id=eq.' + encodeURIComponent(CFG.stateId), {
      method: 'PATCH', body: { desired_state: desired, status: desired === 'online' ? 'starting' : 'stopping' } });
    return json({ ok: true, state: rows && rows[0] });
  }

  if (p === '/api/conversations' && m === 'GET') {
    const rows = await sb(CFG, 'chat_messages?select=conversation_id,role,content,created_at&order=created_at.asc&limit=1000') || [];
    const seen = new Map();
    for (const row of rows) {
      if (!row.conversation_id || row.conversation_id === 'riva-super') continue;
      const s = seen.get(row.conversation_id);
      if (!s) seen.set(row.conversation_id, { id: row.conversation_id, title: row.role === 'user' ? String(row.content || 'New chat').slice(0, 80) : 'Chat', created_at: row.created_at });
      else if (s.title === 'Chat' && row.role === 'user') s.title = String(row.content || 'New chat').slice(0, 80);
    }
    return json([...seen.values()].sort((a, b) => new Date(b.created_at) - new Date(a.created_at)));
  }

  if (p === '/api/conversations' && m === 'POST') {
    const body = await request.json().catch(() => ({}));
    return json({ id: crypto.randomUUID(), title: String(body.title || 'New chat').slice(0, 120), created_at: new Date().toISOString() }, 201);
  }

  if (p.startsWith('/api/conversations/') && p.endsWith('/messages') && m === 'GET') {
    const id = p.split('/')[3];
    const rows = await sb(CFG, 'chat_messages?select=id,conversation_id,role,content,created_at&conversation_id=eq.' + encodeURIComponent(id) + '&order=created_at.asc&limit=200') || [];
    return json(rows.map(r => ({ ...r, sender: r.role })));
  }

  // NEW: delete a chat
  if (p.startsWith('/api/conversations/') && m === 'DELETE') {
    const id = p.split('/')[3];
    if (!id) return json({ error: 'id required' }, 400);
    await sb(CFG, 'chat_messages?conversation_id=eq.' + encodeURIComponent(id), { method: 'DELETE' });
    return json({ ok: true });
  }

  if (p === '/api/model' && m === 'POST') {
    const body = await request.json().catch(() => ({}));
    if (!body.conversation_id) return json({ error: 'conversation_id required' }, 400);
    const sel = String(body.model || 'auto').toLowerCase();
    if (!['auto', '7b', '14b'].includes(sel)) return json({ error: 'model must be auto, 7b, or 14b' }, 400);
    const st = await stateRow();
    if (!st.current_url) return json({ error: 'RIVA is offline. No Gradio URL is available.' }, 503);
    const reply = await runGradioCommand(st.current_url, '/model ' + sel, CFG, body.conversation_id);
    if (!reply.ok) return json({ error: reply.error || 'Model selection failed.' }, 502);
    return json({ ok: true, model: sel, reply: reply.reply || '' });
  }

  if (p === '/api/chat' && m === 'POST') {
    const body = await request.json().catch(() => ({}));
    if (!body.conversation_id || !body.message) return json({ error: 'conversation_id & message required' }, 400);
    const st = await stateRow();
    return await streamGradioToBrowser(st.current_url, body.message, CFG, body.conversation_id, body.model || 'auto');
  }

  if (p === '/api/stats' && m === 'GET') {
    const count = async (table) => {
      try {
        const r = await fetch(CFG.supabaseUrl + '/rest/v1/' + table + '?select=id&limit=1', {
          headers: { apikey: CFG.supabaseKey, Authorization: 'Bearer ' + CFG.supabaseKey, Prefer: 'count=exact', 'Range-Unit': 'items', Range: '0-0' } });
        return parseInt((r.headers.get('content-range') || '').split('/')[1] || '0', 10) || 0;
      } catch { return 0; }
    };
    const [memories, facts, chunks, msgs] = await Promise.all(['memories', 'facts', 'document_chunks', 'chat_messages'].map(count));
    return json({ memories, facts, chunks, msgs });
  }

  return json({ error: 'not found' }, 404);
}
